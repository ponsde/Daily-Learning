#include <bits/stdc++.h>
 
using namespace std;
 
const int N = 60;
int vis[N][N];
int grid[N][N];
 
int dx[] = {0, -1, 0, 1};
int dy[] = {-1, 0, 1, 0};
 
int main(void)
{
    int m, n;
    cin >> m >> n;
    for (int i = 0; i < m; ++i)
    {
        for (int j = 0; j < n; ++j)
        {
            int a;
            cin >> a;
            grid[i][j] = a;
        }
    }
    auto iswall = [&](int x, int y, int i)
    {
        if (x < 0 || x >= m || y < 0 || y >= n)
        {
            return true;
        }
        if ((grid[x][y] >> i) & 1)
        {
            return true;
        }
        return false;
    };
    auto cout_sq = [&](int x0, int y0)
    {
        queue<pair<int, int>> q;
        q.push({x0, y0});
        int ans = 0;
        while (!q.empty())
        {
            auto mm = q.front(); q.pop();
            int x = mm.first;
            int y = mm.second;
            for (int i = 0; i < 4; ++i)
            {
                if (iswall(x, y, i))
                {
                    continue;
                }
                int x1 = x + dx[i];
                int y1 = y + dy[i];
 
                if (vis[x1][y1] == 1)
                {
                    continue;
                }
                vis[x1][y1] = 1;
                q.push({x1, y1});
                ans++;
            }
        }
        return ans;
 
    };
    int cnt = 0;
    int mx_ans = 0;
    for (int i = 0; i < m; ++i)
    {
        for (int j = 0; j < n; ++j)
        {
            if (vis[i][j] == 0)
            {
                mx_ans = max(mx_ans, cout_sq(i, j));
                cnt++;
            }
        }
    }
    cout << cnt << '\n' << mx_ans << '\n';
}
